153template<typename Value_>
154void compare_vectors(const std::vector<Value_>& expected, const std::vector<Value_>& observed, const std::string& context) {
155 const auto n_expected = expected.size();
156 ASSERT_EQ(n_expected, observed.size()) << "mismatch in vector length (" << context << ")";
157 for (I<decltype(n_expected)> i = 0; i < n_expected; ++i) {
158 auto expected_val = expected[i], observed_val = observed[i];
159 if (std::isnan(expected_val)) {
160 ASSERT_EQ(std::isnan(expected_val), std::isnan(observed_val)) << "mismatching NaNs at position " << i << " (" << context << ")";
161 } else {
162 ASSERT_EQ(expected_val, observed_val) << "different values at position " << i << " (" << context << ")";
163 }
164 }
165}

172template<typename Index_>
173std::vector<Index_> simulate_test_access_sequence(const Index_ NR, const Index_ NC, bool use_row, const TestAccessOptions& options) {
174 RngEngine rng(create_seed(NR, NC, use_row, options));
175
176 Index_ start = rng() % options.jump;
177 std::vector<Index_> sequence;
178 const auto limit = (use_row ? NR : NC);
179 if (start < limit) {
180 while (1) {
181 sequence.push_back(start);
182 const Index_ remainder = limit - start;
183 // Make sure this comparison involves two unsigned integers to avoid GCC warnings.
184 if (sanisizer::is_less_than_or_equal(remainder, options.jump)) {
185 break;
186 }
187 start += options.jump;
188 }
189 }
190
191 if (options.order == TestAccessOrder::REVERSE) {
192 std::reverse(sequence.begin(), sequence.end());
193 } else if (options.order == TestAccessOrder::RANDOM) {
194 std::shuffle(sequence.begin(), sequence.end(), rng);
195 }
196
197 return sequence;
198}

200template<bool use_oracle_, typename Index_>
201tatami::MaybeOracle<use_oracle_, Index_> create_oracle(const std::vector<Index_>& sequence, const TestAccessOptions& options) {
202 if constexpr(use_oracle_) {
203 std::shared_ptr<tatami::Oracle<Index_> > oracle;
204 if (options.jump == 1 && options.order == TestAccessOrder::FORWARD) {
205 oracle.reset(new tatami::ConsecutiveOracle<Index_>(0, sanisizer::Cast(sequence.size())));
206 } else {
207 oracle.reset(new tatami::FixedViewOracle<Index_>(sequence.data(), sanisizer::Cast(sequence.size())));
208 }
209 return oracle;
210 } else {
211 return false;
212 }

215template<bool use_oracle_, typename Value_, typename Index_, class SparseExpand_, typename ...Args_>
216void test_access_base(
217 const tatami::Matrix<Value_, Index_>& matrix,
218 const tatami::Matrix<Value_, Index_>& reference,
219 bool use_row,
220 const TestAccessOptions& options,
221 const Index_ extent,
222 const SparseExpand_ sparse_expand,
223 Args_... args
224) {
225 const auto NR = matrix.nrow();
226 ASSERT_EQ(NR, reference.nrow());
227 const auto NC = matrix.ncol();
228 ASSERT_EQ(NC, reference.ncol());
229
230 auto refwork = reference.dense(use_row, args..., tatami::Options());
231
232 auto sequence = simulate_test_access_sequence(NR, NC, use_row, options);
233 auto oracle = create_oracle<use_oracle_>(sequence, options);
234 auto pwork = tatami::make_Extractor<false, use_oracle_>(matrix, use_row, oracle, args..., tatami::Options());
235
236 tatami::Options opt;
237 auto swork = tatami::make_Extractor<true, use_oracle_>(matrix, use_row, oracle, args..., opt);
238 opt.sparse_extract_index = false;
239 auto swork_v = tatami::make_Extractor<true, use_oracle_>(matrix, use_row, oracle, args..., opt);
240 opt.sparse_extract_value = false;
241 auto swork_n = tatami::make_Extractor<true, use_oracle_>(matrix, use_row, oracle, args..., opt);
242 opt.sparse_extract_index = true;
243 auto swork_i = tatami::make_Extractor<true, use_oracle_>(matrix, use_row, oracle, args..., opt);
244
245 sanisizer::as_size_type<std::vector<Value_> >(extent);
246 std::vector<Value_> mat_dense_buffer(extent), ref_dense_buffer(extent), mat_vbuffer(extent), mat_vstore1(extent), mat_vstore2(extent);
247 sanisizer::as_size_type<std::vector<Index_> >(extent);
248 std::vector<Index_> mat_ibuffer(extent), mat_istore1(extent), mat_istore2(extent);
249 bool has_sparse = false;
250
251 for (const auto i : sequence) {
252 {
253 std::fill(ref_dense_buffer.begin(), ref_dense_buffer.end(), 0);
254 const auto ref_buf = ref_dense_buffer.data();
255 const auto ref_ptr = refwork->fetch(i, ref_buf);
256 tatami::copy_n(ref_ptr, extent, ref_buf);
257 }
258
259 // Dense retrieval.
260 {
261 std::fill(mat_dense_buffer.begin(), mat_dense_buffer.end(), 0);
262 const auto mat_buf = mat_dense_buffer.data();
263 const auto mat_ptr = [&]() {
264 if constexpr(use_oracle_) {
265 return pwork->fetch(mat_buf);
266 } else {
267 return pwork->fetch(i, mat_buf);
268 }
269 }();
270 tatami::copy_n(mat_ptr, extent, mat_buf);
271 compare_vectors(ref_dense_buffer, mat_dense_buffer, "dense retrieval");
272 }
273
274 // Sparse retrieval with both values and indices.
275 {
276 std::fill(mat_vbuffer.begin(), mat_vbuffer.end(), 0);
277 std::fill(mat_ibuffer.begin(), mat_ibuffer.end(), 0);
278 const auto vbuf = mat_vbuffer.data();
279 const auto ibuf = mat_ibuffer.data();
280
281 const auto observed = [&]() {
282 if constexpr(use_oracle_) {
283 return swork->fetch(vbuf, ibuf);
284 } else {
285 return swork->fetch(i, vbuf, ibuf);
286 }
287 }();
288 compare_vectors(ref_dense_buffer, sparse_expand(observed), "sparse retrieval");
289
290 if (!has_sparse && sanisizer::is_less_than(observed.number, extent)) {
291 has_sparse = true;
292 }
293
294 bool is_increasing = true;
295 for (I<decltype(observed.number)> i = 1; i < observed.number; ++i) {
296 if (observed.index[i] <= observed.index[i-1]) {
297 is_increasing = false;
298 break;
299 }
300 }
301 ASSERT_TRUE(is_increasing);
302
303 mat_vstore1.clear();
304 mat_vstore1.insert(mat_vstore1.end(), observed.value, observed.value + observed.number);
305 mat_istore1.clear();
306 mat_istore1.insert(mat_istore1.end(), observed.index, observed.index + observed.number);
307 }
308
309 // Sparse retrieval with indices only.
310 {
311 std::fill(mat_ibuffer.begin(), mat_ibuffer.end(), 0);
312 const auto ibuf = mat_ibuffer.data();
313
314 auto observed_i = [&]() {
315 if constexpr(use_oracle_) {
316 return swork_i->fetch(NULL, ibuf);
317 } else {
318 return swork_i->fetch(i, NULL, ibuf);
319 }
320 }();
321
322 ASSERT_TRUE(observed_i.value == NULL);
323 mat_istore2.clear();
324 mat_istore2.insert(mat_istore2.end(), observed_i.index, observed_i.index + observed_i.number);
325 ASSERT_EQ(mat_istore1, mat_istore2);
326 }
327
328 // Sparse retrieval with values only.
329 {
330 std::fill(mat_vbuffer.begin(), mat_vbuffer.end(), 0);
331 const auto vbuf = mat_vbuffer.data();
332
333 auto observed_v = [&]() {
334 if constexpr(use_oracle_) {
335 return swork_v->fetch(vbuf, NULL);
336 } else {
337 return swork_v->fetch(i, vbuf, NULL);
338 }
339 }();
340
341 ASSERT_TRUE(observed_v.index == NULL);
342 mat_vstore2.clear();
343 mat_vstore2.insert(mat_vstore2.end(), observed_v.value, observed_v.value + observed_v.number);
344 compare_vectors(mat_vstore1, mat_vstore2, "sparse retrieval with values only");
345 }
346
347 // Sparse retrieval with neither indices or values.
348 {
349 auto observed_n = [&]() {
350 if constexpr(use_oracle_) {
351 return swork_n->fetch(NULL, NULL);
352 } else {
353 return swork_n->fetch(i, NULL, NULL);
354 }
355 }();
356
357 ASSERT_TRUE(observed_n.value == NULL);
358 ASSERT_TRUE(observed_n.index == NULL);
359 ASSERT_EQ(mat_vstore1.size(), observed_n.number);
360 }
361 }
362
363 if (options.check_sparse && matrix.is_sparse()) {
364 EXPECT_TRUE(has_sparse);
365 }
