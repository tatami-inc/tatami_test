42 auto ptr = new std::vector<Index_>;
43 tatami::VectorPtr<Index_> output(ptr);
44 if (start == max) {
45 return output;
46 }
47 assert(start < max);
48
49 SeedType seed0;

51 seed0 = *seed;

53 seed0 = create_seed(start, max, probability);

57 ptr->push_back(start);
58 std::uniform_real_distribution udist;

65 ptr->push_back(max - 1);
66 }
67

69}
